# 对比函数
def item_comparison(item, detection, epsilon_center=0.1, epsilon_width=0.1, epsilon_height=0.1):
    """
    假设 item 和 detection 有 .center.x、.center.y、.box.width 和 .box.height 属性
    具体的属性请根据实际情况进行修改, 反正就是要计算中心点欧氏距离与框的宽和高
    注意：比较框的宽和高的差距是更好的 (因为不同宽高有可能有相同的面积, 比较面积不太好)
    """
    # 类别必须相同
    if item.class_id != detection.class_id:
        return False

    # 计算中心点欧氏距离
    dx = item.center.x - detection.center.x
    dy = item.center.y - detection.center.y
    center_distance = sqrt(dx * dx + dy * dy)
    if center_distance > epsilon_center:
        return False
    
    # 宽度相对差 (用于归一化)
    max_width = max(item.box.width, detection.box.width)
    if max_width <= 0:
        return False
    # 计算相对宽差
    relative_width_difference = abs(item.box.width - detection.box.width)/ max_width
    if relative_width_difference > epsilon_width:
        return False

    # 高度相对差 (用于归一化)
    max_height = max(item.box.height, detection.box.height)
    if max_height <= 0:
        return False
    # 计算相对高差
    relative_height_difference = abs(item.box.height - detection.box.height) / max_height
    if relative_height_difference > epsilon_height:
        return False

    return True


# 更新函数
def update_item(item, detection, old_count):
    """
    假设 item 和 detection 有 .center.x、.center.y、.box.x、.box.y、.box.width 和 .box.height 属性
    具体的属性请根据实际情况进行修改, 反正就是要将 item 的属性更新为大家的平均值
    """
    new_count = old_count + 1

    item.center.x = (item.center.x * old_count + detection.center.x) / new_count
    item.center.y = (item.center.y * old_count + detection.center.y) / new_count
    
    item.box.x = (item.box.x * old_count + detection.box.x) / new_count
    item.box.y = (item.box.y * old_count + detection.box.y) / new_count

    item.box.width = (item.box.width * old_count + detection.box.width) / new_count
    item.box.height = (item.box.height * old_count + detection.box.height) / new_count


# 合并Yolo的检测结果
def merge_frame_detections(yolo_result, epsilon_center, epsilon_width, epsilon_height
):
    """
    遍历 Yolo 的检测结果, 把[同类物品、中心点距离特别近、框的宽和高也特别近]的两个框合并
    按道理 NMS 会解决这个问题的, 我们这里以防万一再处理一下
    即：手动给 Yolo 搽屁股
    """
    merged_records = []

    for detection in yolo_result:
        matched_record = None

        for record in merged_records:
            if item_comparison(record["item"], detection, epsilon_center, epsilon_width, epsilon_height):
                matched_record = record
                break   # 如果已经匹配到, 则跳出循环

        if matched_record is None:
            merged_records.append({"item": copy(detection), "count": 1})

        else:
            # 同一帧中的重复框取平均
            update_item(
                matched_record["item"],
                detection,
                matched_record["count"]
            )

            matched_record["count"] += 1

    return [
        record["item"]
        for record in merged_records
    ]

# 获取快照函数
def get_snapshot(target_frame_count=3, epsilon_center=0.1, epsilon_width=0.1, epsilon_height=0.1):
    """
    target_frame_count: 超参数, 用 n 个 Yolo 检测结果来合成一个快照
    """
    items_dict = {}          # 存放当前快照中所有的检测到的物品信息 (key=物体信息, value=出现次数)
    count_dict = {}         # 用来计数, 用于求平均
    valid_frame_count = 0    # 当前已经处理的Yolo 检测结果的数量

    # 初步生成快照
    while valid_frame_count < target_frame_count:
        yolo_result = run_yolo()    # 从摄像头中获取一帧的 Yolo 检测结果

        # 首先检查当前画面中是否有"手"
        hand_detected = False

        for detection in yolo_result:
            if detection.class_name == "hand":
                hand_detected = True
                break       # 只要有一个物品是"手", 就跳出循环 (跳过当前帧)

        if hand_detected:   # 如果当前帧的物品有"手", 就重置参数并清空字典 (重新生成快照)
            valid_frame_count = 0
            items_dict.clear()
            continue        # 跳过当前帧, 马上找下一个帧的 Yolo 检测结果

        valid_frame_count += 1

        # 对本轮每一个检测结果寻找已有物品
        for detection in yolo_result:
            matched = False
            for item, count in items_dict.items():
                if item_comparison(item, detection, epsilon_center=epsilon_center, epsilon_width=epsilon_width, epsilon_height=epsilon_height):
                    items_dict[item] = count + 1
                    count_dict[item] += 1
                    update_item(item, detection, count)
                    matched = True
                    break    # 以匹配到一个物品了, 跳过就行

            if not matched:  # 没匹配到就说明是新的物品
                items_dict[detection] = 1
                count_dict[detection] = 0

    # 过滤物品, 得到最后的快照
    items_dict = {key:value 
        for key, value in items_dict.items() if value >= (target_frame_count/2) 
    }
    # 最后的 items_dict 就是一个快照了
    return items_dict
